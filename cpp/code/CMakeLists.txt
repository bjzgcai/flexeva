cmake_minimum_required(VERSION 3.20)

add_library(flexeva_code_layer src/layer.cpp)
add_library(FlexEva::code_layer ALIAS flexeva_code_layer)
target_include_directories(flexeva_code_layer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(flexeva_code_layer PUBLIC cxx_std_20)
target_link_libraries(flexeva_code_layer PUBLIC FlexEva::core)
target_compile_options(flexeva_code_layer PRIVATE -Wall -Wextra -Wpedantic)

add_library(flexeva_code_runtime SHARED src/runtime.cpp)
add_library(FlexEva::code_runtime ALIAS flexeva_code_runtime)
target_include_directories(flexeva_code_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(flexeva_code_runtime PUBLIC cxx_std_20)
target_link_libraries(flexeva_code_runtime PRIVATE ${CMAKE_DL_LIBS})
target_compile_options(flexeva_code_runtime PRIVATE -Wall -Wextra -Wpedantic)

option(FLEXEVA_CODE_BUILD_CPP "Build the Clang host C++ Code frontend" OFF)
if(FLEXEVA_CODE_BUILD_CPP)
    # LLVM's package probes libffi and terminfo with C compile checks.
    enable_language(C)
    find_package(LLVM 14 REQUIRED CONFIG)
    # Ubuntu's Clang 14 package does not ship a ConfigVersion file.
    find_package(Clang REQUIRED CONFIG HINTS "${LLVM_INSTALL_PREFIX}/lib/cmake/clang")
    if(NOT LLVM_VERSION_MAJOR EQUAL 14)
        message(FATAL_ERROR "The host frontend requires the Clang/LLVM 14 toolchain")
    endif()
    add_library(flexeva_code_frontend src/frontend.cpp)
    add_library(FlexEva::code_frontend ALIAS flexeva_code_frontend)
    target_include_directories(flexeva_code_frontend PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_include_directories(flexeva_code_frontend SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
    target_compile_definitions(flexeva_code_frontend PRIVATE ${LLVM_DEFINITIONS})
    target_compile_definitions(flexeva_code_frontend PRIVATE FLEXEVA_CODE_INCLUDE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/include")
    target_compile_features(flexeva_code_frontend PUBLIC cxx_std_20)
    target_link_libraries(flexeva_code_frontend PRIVATE clang-cpp LLVM)
    if(FLEXEVA_BUILD_MAYA)
        add_library(flexeva_code_maya src/maya_adapter.cpp)
        add_library(FlexEva::code_maya ALIAS flexeva_code_maya)
        target_include_directories(flexeva_code_maya PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
        target_include_directories(flexeva_code_maya SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
        target_compile_features(flexeva_code_maya PUBLIC cxx_std_20)
        target_link_libraries(flexeva_code_maya PUBLIC FlexEva::code_runtime FlexEva::maya PRIVATE LLVM)
    endif()
    add_executable(flexeva-code-cpp src/tool.cpp)
    target_link_libraries(flexeva-code-cpp PRIVATE FlexEva::code_frontend)
    target_compile_definitions(flexeva-code-cpp PRIVATE
        FLEXEVA_CODE_INCLUDE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/include"
        FLEXEVA_CODE_RUNTIME_DIR="$<TARGET_FILE_DIR:flexeva_code_runtime>")
    add_dependencies(flexeva-code-cpp flexeva_code_runtime)
endif()
