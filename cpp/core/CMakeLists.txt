cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(flexeva_core VERSION 0.1.0 LANGUAGES CXX)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../ras" ras)
endif()

add_library(flexeva_core src/runtime.cpp src/views.cpp src/codec.cpp src/chunks.cpp src/lineage.cpp src/bundles.cpp src/ground.cpp src/bundle_lifetime.cpp)
add_library(FlexEva::core ALIAS flexeva_core)
set_target_properties(flexeva_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(flexeva_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(flexeva_core PUBLIC FlexEva::ras)
target_compile_features(flexeva_core PUBLIC cxx_std_20)
set_target_properties(flexeva_core PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(flexeva_core PRIVATE -Wall -Wextra -Wpedantic)
