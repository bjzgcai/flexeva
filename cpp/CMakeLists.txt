cmake_minimum_required(VERSION 3.20)
project(flexeva_native LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
option(FLEXEVA_BUILD_CODE "Build the backend-independent Code runtime" ON)
option(FLEXEVA_BUILD_MAYA "Build the Maya evaluator and FakeCUDA" ON)
if(FLEXEVA_BUILD_MAYA AND NOT FLEXEVA_BUILD_CODE)
    message(FATAL_ERROR "The Maya evaluator requires FLEXEVA_BUILD_CODE")
endif()

add_subdirectory(ras)
add_subdirectory(core)
if(FLEXEVA_BUILD_CODE)
    add_subdirectory(code)
endif()
if(FLEXEVA_BUILD_MAYA)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../backends/maya/cpp/fake_cuda
                     ${CMAKE_CURRENT_BINARY_DIR}/fake-cuda)
    add_subdirectory(maya)
endif()
